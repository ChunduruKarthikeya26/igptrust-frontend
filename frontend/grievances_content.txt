import { useState, useEffect } from 'react'
import { useNavigate, Link } from 'react-router-dom'
import { useTranslation } from 'react-i18next'
import { Shield, ArrowLeft, AlertTriangle, CheckCircle, Globe } from 'lucide-react'

const API = import.meta.env.VITE_API_URL || 'http://localhost:8000'

const CATEGORIES = [
  { key: 'erasure_request',   label: 'Data Erasure Request',     desc: 'Request deletion of all your personal data (Section 13)' },
  { key: 'access_request',    label: 'Data Access Request',      desc: 'Request a copy of all personal data held about you (Section 11)' },
  { key: 'correction_request',label: 'Data Correction Request',  desc: 'Request correction of inaccurate personal data' },
  { key: 'consent_issue',     label: 'Consent Issue',            desc: 'Raise an issue about how your consent was collected or used' },
  { key: 'other',             label: 'Other',                    desc: 'Any other grievance related to your personal data' },
]

export default function GrievancePage() {
  const { t } = useTranslation()
  const navigate = useNavigate()
  const [email, setEmail] = useState('')
  const [websites, setWebsites] = useState([])
  const [form, setForm] = useState({ website_id: '', name: '', category: '', description: '' })
  const [submitting, setSubmitting] = useState(false)
  const [submitted, setSubmitted] = useState(null)
  const [error, setError] = useState('')

  useEffect(() => {
    const stored = sessionStorage.getItem('portal_email')
    if (!stored) { navigate('/portal'); return }
    setEmail(stored)
    fetchWebsites(stored)
  }, [])

  const fetchWebsites = async (userEmail) => {
    try {
      const res = await fetch(`${API}/portal/consents?visitor_id=${encodeURIComponent(userEmail)}`)
      if (!res.ok) return
      const data = await res.json()
      const unique = []
      const seen = new Set()
      data.forEach(c => {
        if (!seen.has(c.website_id)) {
          seen.add(c.website_id)
          unique.push({ id: c.website_id, domain: c.domain })
        }
      })
      setWebsites(unique)
    } catch {}
  }

  const handleSubmit = async (e) => {
    e.preventDefault()
    if (!form.website_id || !form.name || !form.category || form.description.length < 20) {
      setError('Please fill all fields. Description must be at least 20 characters.')
      return
    }
    setError('')
    setSubmitting(true)
    try {
      const res = await fetch(`${API}/grievances/`, {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({
          website_id: form.website_id,
          visitor_name: form.name,
          visitor_email: email,
          category: form.category,
          description: form.description,
        }),
      })
      if (!res.ok) throw new Error('Failed')
      const data = await res.json()
      setSubmitted(data)
    } catch {
      setError('Failed to submit grievance. Please try again.')
    } finally {
      setSubmitting(false)
    }
  }

  const inputStyle = {
    width: '100%', fontSize: 14, border: '1px solid #e2e8f0', borderRadius: 8,
    padding: '11px 14px', outline: 'none', boxSizing: 'border-box', color: '#0f172a',
    background: '#fff', fontFamily: 'inherit',
  }
  const labelStyle = { display: 'block', fontSize: 11, fontWeight: 700, color: '#64748b', letterSpacing: '0.07em', textTransform: 'uppercase', marginBottom: 8 }

  if (submitted) {
    return (
      <div style={{ minHeight: '100vh', background: '#f8fafc', display: 'flex', alignItems: 'center', justifyContent: 'center', fontFamily: "'Segoe UI', system-ui, sans-serif" }}>
        <div style={{ background: '#fff', border: '1px solid #e2e8f0', borderRadius: 16, padding: 40, maxWidth: 440, width: '100%', textAlign: 'center' }}>
          <div style={{ width: 56, height: 56, borderRadius: '50%', background: '#dcfce7', display: 'flex', alignItems: 'center', justifyContent: 'center', margin: '0 auto 20px' }}>
            <CheckCircle size={28} color="#16a34a" />
          </div>
          <h2 style={{ fontSize: 20, fontWeight: 700, color: '#0f172a', marginBottom: 8 }}>{t('portal.grievance_submitted')}</h2>
          <p style={{ fontSize: 14, color: '#64748b', marginBottom: 24 }}>{t('portal.grievance_submitted_desc')}</p>
          {submitted.reference_number && (
            <div style={{ background: '#f8fafc', border: '1px solid #e2e8f0', borderRadius: 10, padding: '14px 20px', marginBottom: 24 }}>
              <p style={{ fontSize: 11, color: '#94a3b8', textTransform: 'uppercase', letterSpacing: '0.06em', marginBottom: 4 }}>{t('portal.reference_number')}</p>
              <p style={{ fontSize: 16, fontWeight: 700, color: '#0f172a', fontFamily: 'monospace' }}>{submitted.reference_number}</p>
            </div>
          )}
          <div style={{ display: 'flex', gap: 10 }}>
            <button
              onClick={() => { setSubmitted(null); setForm({ website_id: '', name: '', category: '', description: '' }) }}
              style={{ flex: 1, padding: '10px', borderRadius: 8, fontSize: 13, fontWeight: 500, color: '#475569', border: '1px solid #e2e8f0', background: '#fff', cursor: 'pointer' }}
            >
              {t('portal.submit_another')}
            </button>
            <button
              onClick={() => navigate('/portal/dashboard')}
              style={{ flex: 1, padding: '10px', borderRadius: 8, fontSize: 13, fontWeight: 600, color: '#fff', background: '#3b82f6', border: 'none', cursor: 'pointer' }}
            >
              {t('portal.back_to_dashboard')}
            </button>
          </div>
        </div>
      </div>
    )
  }

  return (
    <div style={{ minHeight: '100vh', background: '#f8fafc', fontFamily: "'Segoe UI', system-ui, sans-serif" }}>
      {/* Header */}
      <header style={{ background: '#fff', borderBottom: '1px solid #e2e8f0', position: 'sticky', top: 0, zIndex: 10 }}>
        <div style={{ maxWidth: 600, margin: '0 auto', padding: '12px 20px', display: 'flex', alignItems: 'center', justifyContent: 'space-between' }}>
          <div style={{ display: 'flex', alignItems: 'center', gap: 8 }}>
            <div style={{ width: 28, height: 28, borderRadius: 8, background: '#3b82f6', display: 'flex', alignItems: 'center', justifyContent: 'center' }}>
              <Shield size={14} color="#fff" />
            </div>
            <span style={{ fontSize: 15, fontWeight: 700, color: '#0f172a' }}>{t('portal.grievance_title')}</span>
          </div>
          <Link
            to="/portal/dashboard"
            style={{ display: 'inline-flex', alignItems: 'center', gap: 6, fontSize: 13, fontWeight: 500, color: '#64748b', textDecoration: 'none' }}
          >
            <ArrowLeft size={14} />
            {t('portal.back_to_dashboard')}
          </Link>
        </div>
      </header>

      <main style={{ maxWidth: 600, margin: '0 auto', padding: '28px 20px' }}>
        {/* Rights Banner */}
        <div style={{ background: '#fffbeb', border: '1px solid #fde68a', borderRadius: 12, padding: '14px 18px', marginBottom: 28, display: 'flex', gap: 12 }}>
          <AlertTriangle size={16} color="#d97706" style={{ flexShrink: 0, marginTop: 1 }} />
          <div>
            <p style={{ fontSize: 13, fontWeight: 600, color: '#92400e', marginBottom: 4 }}>{t('portal.your_rights')}</p>
            <p style={{ fontSize: 12, color: '#b45309', lineHeight: 1.6, margin: 0 }}>{t('portal.rights_desc')}</p>
          </div>
        </div>

        {/* Form card */}
        <div style={{ background: '#fff', border: '1px solid #e2e8f0', borderRadius: 16, overflow: 'hidden' }}>
          {/* Gradient bar */}
          <div style={{ height: 4, background: 'linear-gradient(90deg, #3b82f6, #8b5cf6, #ec4899)' }} />

          <form onSubmit={handleSubmit} style={{ padding: '28px 28px 32px' }}>
            {error && (
              <div style={{ display: 'flex', alignItems: 'center', gap: 8, padding: '10px 14px', background: '#fef2f2', border: '1px solid #fecaca', borderRadius: 8, marginBottom: 20, fontSize: 13, color: '#dc2626' }}>
                <AlertTriangle size={13} />
                {error}
              </div>
            )}

            {/* Website */}
            <div style={{ marginBottom: 22 }}>
              <label style={labelStyle}>{t('portal.website_label')} *</label>
              <div style={{ position: 'relative' }}>
                <Globe size={14} color="#94a3b8" style={{ position: 'absolute', left: 12, top: '50%', transform: 'translateY(-50%)' }} />
                <select
                  value={form.website_id}
                  onChange={e => setForm(f => ({ ...f, website_id: e.target.value }))}
                  style={{ ...inputStyle, paddingLeft: 36, appearance: 'none', cursor: 'pointer', color: form.website_id ? '#0f172a' : '#94a3b8' }}
                >
                  <option value="">Select the website this is about</option>
                  {websites.map(w => (
                    <option key={w.id} value={w.id}>{w.domain}</option>
                  ))}
                </select>
              </div>
            </div>

            {/* Name */}
            <div style={{ marginBottom: 22 }}>
              <label style={labelStyle}>{t('portal.name_label')} *</label>
              <input
                type="text"
                value={form.name}
                onChange={e => setForm(f => ({ ...f, name: e.target.value }))}
                placeholder="e.g. Ananya Krishnan"
                style={inputStyle}
              />
            </div>

            {/* Email (read-only) */}
            <div style={{ marginBottom: 22 }}>
              <label style={labelStyle}>{t('portal.email_label')} *</label>
              <input type="email" value={email} readOnly style={{ ...inputStyle, background: '#f8fafc', color: '#64748b' }} />
            </div>

            {/* Category */}
            <div style={{ marginBottom: 22 }}>
              <label style={labelStyle}>{t('portal.category_label')} *</label>
              <div style={{ display: 'flex', flexDirection: 'column', gap: 8 }}>
                {CATEGORIES.map(cat => (
                  <label
                    key={cat.key}
                    style={{
                      display: 'flex', alignItems: 'flex-start', gap: 12, padding: '12px 16px',
                      border: `1px solid ${form.category === cat.key ? '#bfdbfe' : '#e2e8f0'}`,
                      borderRadius: 10, cursor: 'pointer',
                      background: form.category === cat.key ? '#eff6ff' : '#fff',
                      transition: 'all 0.15s',
                    }}
                  >
                    <input
                      type="radio"
                      name="category"
                      value={cat.key}
                      checked={form.category === cat.key}
                      onChange={e => setForm(f => ({ ...f, category: e.target.value }))}
                      style={{ marginTop: 2, accentColor: '#3b82f6' }}
                    />
                    <div>
                      <p style={{ fontSize: 13, fontWeight: 600, color: '#0f172a', margin: 0 }}>{cat.label}</p>
                      <p style={{ fontSize: 12, color: '#64748b', margin: '2px 0 0' }}>{cat.desc}</p>
                    </div>
                  </label>
                ))}
              </div>
            </div>

            {/* Description */}
            <div style={{ marginBottom: 28 }}>
              <label style={labelStyle}>{t('portal.description_label')} *</label>
              <textarea
                value={form.description}
                onChange={e => setForm(f => ({ ...f, description: e.target.value }))}
                rows={5}
                placeholder="Please describe your grievance in detail. Include any relevant dates, data types, or specific concerns..."
                style={{ ...inputStyle, resize: 'vertical', lineHeight: 1.6 }}
              />
              <div style={{ display: 'flex', justifyContent: 'space-between', marginTop: 6 }}>
                {form.description.length > 0 && form.description.length < 20 && (
                  <span style={{ fontSize: 11, color: '#ef4444' }}>Minimum 20 characters</span>
                )}
                <span style={{ fontSize: 11, color: '#94a3b8', marginLeft: 'auto' }}>{form.description.length} characters</span>
              </div>
            </div>

            <button
              type="submit"
              disabled={submitting}
              style={{
                width: '100%', padding: '13px', borderRadius: 10, fontSize: 14, fontWeight: 600,
                color: '#fff', background: submitting ? '#93c5fd' : '#3b82f6',
                border: 'none', cursor: submitting ? 'not-allowed' : 'pointer',
                transition: 'background 0.15s',
              }}
            >
              {submitting ? t('common.loading') : t('portal.submit_grievance')}
            </button>
          </form>
        </div>
      </main>
    </div>
  )
}
